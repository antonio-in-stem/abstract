Example

Product catalog.

A useful first schema because it has scalar fields, defaults, a nested owner object, list fields, and tagged capability objects.

schema Product {
    id: text(1..40)
    name: text(1..80)
    status: enum(draft, active, retired)
    tier: enum(free, standard, premium)
    release_year: int(2000..2100)
    tags[]: enum(core, experimental, internal, public, ai_ready) @optional

    owner {
        team: text(1..50)
        contact: text(1..80)
    }

    capabilities[] {
        id: enum(search, sync, export, audit, automation, analytics) @tag
        availability: enum(alpha, beta, stable) = stable
    }
}
Product :: @id.atlas, @status.active, @tier.premium, @release_year.2026
    name: Atlas Search
    tags: [core, public, ai_ready]
    owner.team: Knowledge Systems
    owner.contact: systems@example.com
    capabilities: [#search, #sync(availability: beta), #analytics]
